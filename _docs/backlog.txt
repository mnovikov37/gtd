  });

  /* Стрелка по умолчанию (закрыто) */
  .menu-toggle::before {
    content: "◀";
    margin-right: 5px;
    transition: transform 0.2s ease; /* Плавный поворот */
    display: inline-block;
  }

  /* Изменение состояния через поворот (или смену символа) */
  .menu-toggle[aria-expanded="true"]::before {
    transform: rotate(-90deg); /* Поворачивает треугольник вниз */
